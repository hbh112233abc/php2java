## 11. 集成logback

官网: https://logback.qos.ch


### 11.1 引入依赖

```xml
<dependency>
    <groupId>net.logstash.logback</groupId>
    <artifactId>logstash-logback-encoder</artifactId>
    <version>9.0</version>
    <scope>runtime</scope> <!-- 如果无需在编译期使用其API，可用runtime -->
</dependency>
```


### 11.2 配置

`demo/src/main/resources/logback-spring.xml`

```xml
<?xml version="1.0" encoding="UTF-8"?>
<configuration>
    <include resource="org/springframework/boot/logging/logback/defaults.xml"/>

    <appender name="CONSOLE" class="ch.qos.logback.core.ConsoleAppender">
        <encoder class="net.logstash.logback.encoder.LogstashEncoder">
            <!-- 可选：添加自定义字段，如应用名和环境 -->
            <customFields>{"app":"${spring.application.name:-my-app}","env":"${spring.profiles.active:-default}"}</customFields>
        </encoder>
    </appender>

    <root level="INFO">
        <appender-ref ref="CONSOLE"/>
    </root>
</configuration>
```

> 这个配置会让所有日志以单行 JSON 格式输出，便于后续的 ELK 或 Loki 等系统采集

进阶配置,精细地控制输出字段，可以使用 `LoggingEventCompositeJsonEncoder` 并搭配 `providers`

添加依赖:`logstash-logback-encoder`

```xml
<!-- logstash JSON 日志编码器，logback-spring.xml 中 LoggingEventCompositeJsonEncoder 依赖此包 -->
<dependency>
    <groupId>net.logstash.logback</groupId>
    <artifactId>logstash-logback-encoder</artifactId>
    <version>7.4</version>
</dependency>
```

修改配置:

```xml
<?xml version="1.0" encoding="UTF-8"?>
<configuration>
    <include resource="org/springframework/boot/logging/logback/defaults.xml"/>

    <!-- 把 Spring 环境属性显式映射成 logback 变量，确保在 logstash 的 pattern 中可解析 -->
    <springProperty name="APP_NAME" source="spring.application.name" defaultValue="my-app"/>
    <springProperty name="APP_ENV" source="spring.profiles.active" defaultValue="default"/>

    <appender name="CONSOLE" class="ch.qos.logback.core.ConsoleAppender">
        <encoder class="net.logstash.logback.encoder.LoggingEventCompositeJsonEncoder">
            <providers>
                <timestamp/>
                <version/>
                <context/>
                <pattern>
                    <pattern>
                        {
                            "level": "%level",
                            "service": "${APP_NAME}",
                            "env": "${APP_ENV}",
                            "traceId": "%X{traceId:-}",
                            "spanId": "%X{spanId:-}",
                            "requestId": "%X{requestId:-}",
                            "message": "%message",
                            "logger": "%logger",
                            "thread": "%thread"
                        }
                    </pattern>
                </pattern>
                <stackTrace/>
            </providers>
        </encoder>
    </appender>

    <root level="INFO">
        <appender-ref ref="CONSOLE"/>
    </root>
</configuration>
```

### 11.3 注入上下文信息

结构化日志的强大之处在于可以自动携带请求上下文。你可以通过 MDC (Mapped Diagnostic Context) 来动态注入业务字段，如 `requestId`,`userId` 等

创建`Servlet Filter`来注入请求ID(requestId)

```java
package com.example.demo.common;

import java.io.IOException;
import java.util.UUID;

import org.slf4j.MDC;
import org.springframework.stereotype.Component;

import jakarta.servlet.Filter;
import jakarta.servlet.FilterChain;
import jakarta.servlet.ServletException;
import jakarta.servlet.ServletRequest;
import jakarta.servlet.ServletResponse;

@Component
public class RequestIdFilter  implements  Filter {
    @Override
    public void doFilter(ServletRequest request, ServletResponse response, FilterChain chain) throws IOException,ServletException {
        String traceId = UUID.randomUUID().toString();
        String requestId = UUID.randomUUID().toString();
        MDC.put("traceId", traceId);
        MDC.put("requestId", requestId);
        try {
            chain.doFilter(request, response);
        } finally {
            MDC.clear(); // 必须清理，避免线程复用污染
        }
    }
}
```

终端日志格式:

```json
{"@timestamp":"2026-09-23T09:03:38.9744842+08:00","@version":"1","level":"INFO","service":"php2java-v0.0.1","env":"dev","traceId":"ec6e44a5-3aec-4ed7-b631-8272dfc0c324","spanId":"","requestId":"ba33ed7f-cc7d-4e12-bc3d-2cdf07da71f0","message":"HikariPool-1 - Added connection com.mysql.cj.jdbc.ConnectionImpl@77e78ef8","logger":"com.zaxxer.hikari.pool.HikariPool","thread":"http-nio-8080-exec-1"}
```

### 11.4 支持spanId(可选)

#### 11.4.1 引入依赖

```xml
<!-- 链路追踪 -->
<dependency>
    <groupId>org.springframework.boot</groupId>
    <artifactId>spring-boot-starter-actuator</artifactId>
</dependency>
<!--
  Spring Boot 4 已彻底移除 Micrometer Tracing / Observation 的自动配置，
  改用 Brave 原生方案（见 BraveTracingConfig）：
  1) brave-context-slf4j 提供 MDCScopeDecorator，把 traceId/spanId 写入 SLF4J MDC
  2) 自定义 Jakarta 过滤器为每个 HTTP 请求建 Span（brave-instrumentation-servlet
     使用 javax.servlet，与 SB4 的 jakarta.servlet 不兼容，故不用它）
-->
<dependency>
    <groupId>io.zipkin.brave</groupId>
    <artifactId>brave-context-slf4j</artifactId>
    <version>6.3.0</version>
</dependency>
<!-- Zipkin 上报（brave 原生 reporter；如需真正上报，见 BraveTracingConfig 中的注释） -->
<dependency>
    <groupId>io.zipkin.reporter2</groupId>
    <artifactId>zipkin-reporter-brave</artifactId>
</dependency>
```

#### 11.4.2 配置类

```java
package com.example.demo.common;

import java.io.IOException;

import jakarta.servlet.Filter;
import jakarta.servlet.FilterChain;
import jakarta.servlet.ServletException;
import jakarta.servlet.ServletRequest;
import jakarta.servlet.ServletResponse;
import jakarta.servlet.http.HttpServletRequest;

import org.springframework.beans.factory.annotation.Value;
import org.springframework.boot.web.servlet.FilterRegistrationBean;
import org.springframework.context.annotation.Bean;
import org.springframework.context.annotation.Configuration;

import brave.Span;
import brave.Tracer;
import brave.Tracing;
import brave.context.slf4j.MDCScopeDecorator;
import brave.propagation.CurrentTraceContext;
import brave.propagation.ThreadLocalCurrentTraceContext;

/**
 * Spring Boot 4 已移除 Micrometer Tracing / Observation 的自动配置，
 * 这里改用 Brave 原生方式实现链路追踪：
 * <ul>
 *   <li>{@link MDCScopeDecorator#get()} 负责把 traceId / spanId 写入 SLF4J 的 MDC，
 *       从而被 logback 的 %X{traceId} / %X{spanId} 读取到日志中；</li>
 *   <li>{@link BraveRequestFilter}（Jakarta 过滤器）为每个 HTTP 请求自动创建 Span 并开启 MDC 作用域。
 *       注意：brave-instrumentation-servlet 用的是 javax.servlet，与 SB4 的 jakarta.servlet 不兼容，故不用它。</li>
 * </ul>
 */
@Configuration
public class BraveTracingConfig {
    @Value("${spring.application.name}")
    private String appName;

    /** 原生 Brave Tracing：开启 MDC 装饰器，使 traceId/spanId 进入 MDC */
    @Bean
    public Tracing tracing() {
        // 关键：MDCScopeDecorator 让 traceId/spanId 进入 MDC
        CurrentTraceContext currentTraceContext = ThreadLocalCurrentTraceContext.newBuilder()
            .addScopeDecorator(MDCScopeDecorator.get())
            .build();

        return Tracing.newBuilder()
            .localServiceName(appName)
            .currentTraceContext(currentTraceContext)
            // 如需上报 Zipkin，先引入依赖 io.zipkin.reporter2:zipkin-reporter-urlconnection
            // （版本需与 zipkin-reporter 保持一致），再取消下面这行注释：
            // .addSpanHandler(AsyncZipkinSpanHandler.create(
            //     URLConnectionSender.create("http://localhost:9411/api/v2/spans")))
            .build();
    }

    /** 注册自定义过滤器：每个请求自动建立 Span 并写入 MDC */
    @Bean
    public FilterRegistrationBean<BraveRequestFilter> tracingFilter(Tracing tracing) {
        FilterRegistrationBean<BraveRequestFilter> registration = new FilterRegistrationBean<>();
        registration.setFilter(new BraveRequestFilter(tracing.tracer()));
        registration.addUrlPatterns("/*");
        // 放在最前面，保证请求一开始（进入 Controller 之前）就建立 Span
        registration.setOrder(Integer.MIN_VALUE);
        return registration;
    }

    /** 每个请求创建 Brave Span，并通过 withSpanInScope 触发 MDCScopeDecorator 写入 MDC */
    public static class BraveRequestFilter implements Filter {

        private final Tracer tracer;

        public BraveRequestFilter(Tracer tracer) {
            this.tracer = tracer;
        }

        @Override
        public void doFilter(ServletRequest request, ServletResponse response, FilterChain chain)
                throws IOException, ServletException {
            String name = request instanceof HttpServletRequest httpRequest
                ? httpRequest.getMethod() + " " + httpRequest.getRequestURI()
                : "unknown";
            Span span = tracer.nextSpan().name(name).start();
            // withSpanInScope 会进入 MDC 作用域，traceId/spanId 被写入 SLF4J MDC
            try (Tracer.SpanInScope ignored = tracer.withSpanInScope(span)) {
                chain.doFilter(request, response);
            } finally {
                span.finish();
            }
        }
    }
}
```

#### 11.4.3 相关配置

- `application-dev.yml`配置日志级别

```yml
logging:
  level:
    com.example.demo.mapper: debug   # 换成你实际的 mapper 包路径
```

- `application-dev.yml`配置`MyBatis-Plus`日志输出使用`Sl4jImpl`

```yml
# MyBatis-Plus 配置 (可选)
mybatis-plus:
  configuration:
    # log-impl: org.apache.ibatis.logging.stdout.StdOutImpl # 开启 SQL 日志，方便调试
    log-impl: org.apache.ibatis.logging.slf4j.Slf4jImpl # 开启 SQL 日志（走 logback，可带 MDC 的 traceId/spanId）
    map-underscore-to-camel-case: true # 默认开启，下划线转驼峰
  mapper-locations: classpath*:/mapper/**/*.xml # 如果使用 XML 文件，指定其位置
```

public class UserController {

        }
        return Result.success(
            userService.page(new Page<>(page, pageSize), q)
        );
    }
    ...
}

### 11.5 日志留存

`demo/src/main/resources/logback-spring.xml` 配置

根据不同环境配置

```xml
<?xml version="1.0" encoding="UTF-8"?>
<configuration>

    <!-- 公共控制台输出（可选） -->
    <appender name="CONSOLE" class="ch.qos.logback.core.ConsoleAppender">
        <encoder>
            <pattern>%d{yyyy-MM-dd HH:mm:ss.SSS} [%thread] %-5level %logger{36} - %msg%n</pattern>
        </encoder>
    </appender>

    <!-- ========== 开发环境：控制台输出 + 本地文件，DEBUG 级别 ========== -->
    <springProfile name="dev">
        <appender name="DEV_FILE" class="ch.qos.logback.core.rolling.RollingFileAppender">
            <file>logs/app.log</file>
            <rollingPolicy class="ch.qos.logback.core.rolling.TimeBasedRollingPolicy">
                <fileNamePattern>logs/app.%d{yyyy-MM-dd}.log</fileNamePattern>
                <maxHistory>7</maxHistory>
            </rollingPolicy>
            <encoder>
                <pattern>%d{HH:mm:ss.SSS} [%thread] %-5level %logger{36} - %msg%n</pattern>
            </encoder>
        </appender>

        <root level="DEBUG">
            <appender-ref ref="CONSOLE"/>
            <appender-ref ref="DEV_FILE"/>
        </root>
    </springProfile>

    <!-- ========== 生产环境：仅文件输出，INFO 级别 ========== -->
    <springProfile name="prod">
        <appender name="PROD_FILE" class="ch.qos.logback.core.rolling.RollingFileAppender">
            <file>logs/app.log</file>
            <rollingPolicy class="ch.qos.logback.core.rolling.TimeBasedRollingPolicy">
                <fileNamePattern>logs/app.%d{yyyy-MM-dd}.log</fileNamePattern>
                <maxHistory>30</maxHistory>
            </rollingPolicy>
            <encoder>
                <pattern>%d{yyyy-MM-dd HH:mm:ss.SSS} [%thread] %-5level %logger{36} - %msg%n</pattern>
            </encoder>
        </appender>

        <root level="INFO">
            <appender-ref ref="PROD_FILE"/>
        </root>
    </springProfile>
</configuration>
```
